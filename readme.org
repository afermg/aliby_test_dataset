#+TITLE: Generating a small test dataset for aliby
#+AUTHOR: Alán F. Muñoz

* Overview

This repository is an [[file:.agent-shell/transcripts/][LLM-aided]] effort to generate a small, reproducible dataset for integration testing the [[https://github.com/afermg/aliby][aliby]] framework for high throughput cell microscopy. 

The original full-size images were obtained from the JUMP_lite dataset. This project contains the logic to:
1. Crop those full-size images into smaller 256x256 (cell painting) or 293x293 (time series) crops.
2. Group the resulting TIFF channels into a single `zarr` array using `zstd` compression via the provided Python script.
3. Package the outputs into a single `.tar.gz` archive suitable for integration testing.
4. Upload the dataset directly to [[https://zenodo.org/records/19228474][Zenodo]] to act as an external test artifact for CI workflows.

* Download original data

The full dataset is hosted on Zenodo. We first download and extract it to ensure the pipeline is reproducible from scratch.
#+begin_src sh
  mkdir -p data
  curl -L -o data/aliby_full_dataset.tar.gz "https://zenodo.org/api/records/19410594/files/aliby_full_dataset.tar.gz/content"
  cd data
  tar -xzvf aliby_full_dataset.tar.gz
  rm aliby_full_dataset.tar.gz
  cd ..
#+end_src

*  Crop images
** Reduce the size of the images
#+begin_src sh
  cd data
  # Cell painting
  mkdir -p crop_cellpainting_256
  for file in full_cellpainting_1080/*.tif; do
      filename=$(basename "$file")
      magick "$file" -gravity center -crop 256x256+0+0 +repage -compress Zstd "crop_cellpainting_256/$filename"
  done
  
  # Time series
  for ds in full_timeseries_*_1172; do
      if [ -d "$ds" ]; then
          out_dir="crop_${ds#full_}"
          out_dir="${out_dir/_1172/_293}"
          mkdir -p "$out_dir"
          for dir in "$ds"/*; do
              if [ -d "$dir" ]; then
                  dirname=$(basename "$dir")
                  mkdir -p "$out_dir/$dirname"
                  for file in "$dir"/*.tif; do
                      filename=$(basename "$file")
                      magick "$file" -gravity center -crop 293x293+0+0 +repage -compress Zstd "$out_dir/$dirname/$filename"
                  done
              fi
          done
      fi
  done
  cd ..
#+end_src

** Convert folder of tiffs into a zarr dataset
#+begin_src sh :results raw :wrap example
  # Cell painting
  uv run tiff_to_zarr.py data/crop_cellpainting_256 data/crop_cellpainting_256.zarr -n 5
  
  # Time series
  for ds in data/crop_timeseries_*_293; do
      if [ -d "$ds" ]; then
          dsname=$(basename "$ds" | sed 's/^crop_//; s/_293$//')
          out_zarr_dir="data/crop_${dsname}_293_zarrs"
          mkdir -p "$out_zarr_dir"
          for dir in "$ds"/*; do
              if [ -d "$dir" ]; then
                  dirname=$(basename "$dir")
                  uv run timeseries_to_zarr.py "$dir" "$out_zarr_dir/${dirname}.zarr"
              fi
          done
      fi
  done
#+end_src
